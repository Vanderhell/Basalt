cmake_minimum_required(VERSION 3.16)
project(BasaltDB C)

find_package(Threads REQUIRED)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)

add_library(BasaltDB STATIC
    src/jobdb.c
    src/jobdb_codec.c
    src/jobdb_crc32c.c
    src/jobdb_file.c
    src/jobdb_lock.c
)
target_include_directories(BasaltDB PUBLIC include PRIVATE src)
if (MSVC)
  target_compile_definitions(BasaltDB PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(BasaltDB PRIVATE /W4 /WX)
else()
  target_compile_options(BasaltDB PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

enable_testing()
add_executable(BasaltDB_tests tests/test_jobdb.c)
target_link_libraries(BasaltDB_tests PRIVATE BasaltDB)
if (MSVC)
  target_compile_options(BasaltDB_tests PRIVATE /UNDEBUG)
else()
  target_compile_options(BasaltDB_tests PRIVATE -UNDEBUG)
endif()
if (NOT MSVC)
  target_link_libraries(BasaltDB_tests PRIVATE m)
endif()
add_test(NAME BasaltDB_tests COMMAND BasaltDB_tests)

add_executable(BasaltDB_foundation tests/test_jobdb_foundation.c)
target_link_libraries(BasaltDB_foundation PRIVATE BasaltDB)
if (MSVC)
  target_compile_options(BasaltDB_foundation PRIVATE /UNDEBUG)
  target_compile_options(BasaltDB_foundation PRIVATE /W4 /WX)
else()
  target_compile_options(BasaltDB_foundation PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME BasaltDB_foundation COMMAND BasaltDB_foundation)

add_executable(BasaltDB_gate tests/test_jobdb_gate.c)

  target_compile_definitions(BasaltDB_gate PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(BasaltDB_gate PRIVATE /UNDEBUG /W4 /WX)
else()
  target_compile_options(BasaltDB_gate PRIVATE -UNDEBUG -Wall -Wextra -Wpedantic -Werror)
endif()
add_test(NAME BasaltDB_wal_recovery COMMAND BasaltDB_gate wal)
add_test(NAME BasaltDB_corruption COMMAND BasaltDB_gate corruption)
add_test(NAME BasaltDB_transactions COMMAND BasaltDB_gate transactions)
add_test(NAME BasaltDB_failure_injection COMMAND BasaltDB_gate failure)
add_test(NAME BasaltDB_stress COMMAND BasaltDB_gate stress)
add_test(NAME BasaltDB_lease COMMAND BasaltDB_gate lease)
add_test(NAME BasaltDB_multiprocess COMMAND BasaltDB_gate multiprocess)

add_library(BasaltCore STATIC src/jobcore.c src/basalt_storage.c)
target_sources(BasaltCore PRIVATE src/jobcore_cron.c)
target_include_directories(BasaltCore PUBLIC include)
target_link_libraries(BasaltCore PUBLIC BasaltDB Threads::Threads)
if (MSVC)
  target_compile_definitions(BasaltCore PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(BasaltCore PRIVATE /W4 /WX)
else()
  target_compile_options(BasaltCore PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()

add_library(basalt_core_shared SHARED src/jobcore.c src/jobcore_cron.c src/basalt_storage.c src/basalt_abi.c)
target_include_directories(basalt_core_shared PUBLIC include)
target_link_libraries(basalt_core_shared PUBLIC BasaltDB Threads::Threads)
if (MSVC)
  target_compile_definitions(basalt_core_shared PRIVATE _CRT_SECURE_NO_WARNINGS)
  target_compile_options(basalt_core_shared PRIVATE /W4 /WX)
  target_link_options(basalt_core_shared PRIVATE /EXPORT:basalt_ledger_get)
  target_link_options(basalt_core_shared PRIVATE /EXPORT:basalt_ledger_list /EXPORT:basalt_schedule_list)

  target_compile_options(basalt_core_shared PRIVATE -Wall -Wextra -Wpedantic -Werror)
